\section{Information-Theoretic Derivations}
\label{app:info_theory}
%=============================================================================

\subsection{Stable Rank, Participation Ratio, and R\'{e}nyi Entropy}

For a weight matrix $W \in \R^{m \times n}$ with singular values $\sigma_1 \geq \sigma_2 \geq \cdots \geq \sigma_r$ ($r = \min(m,n)$), define the spectral probability distribution:
\begin{equation}
    p_i = \frac{\sigma_i^2}{\sum_j \sigma_j^2} = \frac{\sigma_i^2}{\|W\|_F^2}, \qquad \sum_i p_i = 1.
\end{equation}

The R\'{e}nyi entropy of order $q$ is:
\begin{equation}
    H_q(\mathbf{p}) = \frac{1}{1-q} \log\!\left(\sum_i p_i^q\right).
\end{equation}
For $q = 2$ (collision entropy):
\begin{equation}
    H_2 = -\log\!\left(\sum_i p_i^2\right).
\end{equation}

The \emph{participation ratio} (inverse Herfindahl index) is:
\begin{equation}
    \mathrm{PR} = \exp(H_2) = \frac{1}{\sum_i p_i^2} = \frac{\left(\sum_i \sigma_i^2\right)^2}{\sum_i \sigma_i^4} = \frac{\|W\|_F^4}{\sum_i \sigma_i^4}.
\end{equation}

The \emph{stable rank} is:
\begin{equation}
    \mathrm{SR}(W) = \frac{\|W\|_F^2}{\sigma_1^2} = \frac{\sum_i \sigma_i^2}{\sigma_1^2}.
\end{equation}

\paragraph{Inequality chain.} The following hierarchy holds for any matrix:
\begin{equation}
    1 \;\leq\; \mathrm{SR} \;\leq\; \mathrm{PR} = \exp(H_2) \;\leq\; \exp(H_1) \;\leq\; r = \mathrm{rank}(W).
\end{equation}

\paragraph{When does $\mathrm{SR} \approx \mathrm{PR}$?}
$\mathrm{SR} = \mathrm{PR}$ when all non-zero singular values are equal (uniform spectrum). More generally, for a geometric spectrum $\sigma_i^2 = \sigma_1^2 \cdot \rho^{i-1}$ with $0 < \rho < 1$:
\begin{align}
    \mathrm{SR} &= \frac{1-\rho^r}{1-\rho} \cdot \frac{1-\rho}{1-\rho^r} = \frac{1-\rho^r}{1-\rho}, \\
    \mathrm{PR} &= \frac{(1-\rho^r)^2}{(1-\rho^{2r})(1-\rho)} \cdot \frac{1-\rho^2}{1-\rho} \approx \frac{1-\rho}{1-\rho^2} \cdot \mathrm{SR}.
\end{align}
For $\rho < 0.5$ (typical of trained transformers where the spectrum decays rapidly), $\mathrm{PR}/\mathrm{SR} \to (1+\rho)^{-1} \in [0.67, 1.0]$. Thus in the trained regime, SR is a tight lower bound on PR, and $\log(\mathrm{SR}) \approx H_2$ up to a constant $\leq \log(1.5)$.

\subsection{Universal Compression Ratio and Entropy Reduction}

The Universal Compression Ratio is:
\begin{equation}
    \mathrm{UCR} = \frac{\mathrm{SR}_{\mathrm{final}}}{\mathrm{SR}_{\mathrm{init}}} \approx 0.13.
\end{equation}

Since $\mathrm{SR} \approx \exp(H_2)$:
\begin{equation}
    \Delta H_2 = \log(\mathrm{UCR}) = \log(0.13) \approx -2.0 \;\text{nats}.
\end{equation}

At initialization (Marchenko-Pastur), for an $m \times n$ matrix with $m > n$:
\begin{equation}
    \mathrm{SR}_{\mathrm{init}} \approx \frac{mn}{(\sqrt{m} + \sqrt{n})^2} \approx \frac{d}{(1 + \sqrt{a})^2 / a}
\end{equation}
where $a = m/n$ is the aspect ratio. For square matrices ($a = 1$): $\mathrm{SR}_{\mathrm{init}} \approx d/4$. For typical transformer layers ($a \approx 4$): $\mathrm{SR}_{\mathrm{init}} \approx 4d/9 \approx 0.4d$, consistent with the measured $\mathrm{SR}/d_{\mathrm{init}} \approx 0.39$.

\section{Langevin Dynamics and Spectral Steady State}
\label{app:langevin}
%=============================================================================

\subsection{From SGD to Langevin Equation}

Consider SGD with learning rate $\eta$, weight decay $\lambda$, and batch size $B$ on parameter $\theta$:
\begin{equation}
    \theta_{t+1} = \theta_t - \eta \left(\hat{g}_t + \lambda \theta_t\right)
\end{equation}
where $\hat{g}_t = \nabla_\theta \hat{\mathcal{L}}_B(\theta_t)$ is the mini-batch gradient. Decomposing $\hat{g}_t = g(\theta_t) + \xi_t$ (true gradient plus noise with $\E[\xi_t] = 0$, $\mathrm{Cov}[\xi_t] = \Sigma_g / B$), the continuous-time limit is:
\begin{equation}
    d\theta = -\eta(g(\theta) + \lambda\theta)\,dt + \sqrt{\eta}\,\Sigma_g^{1/2}\,dW_t / \sqrt{B}.
\end{equation}

Following~\citet{liu_tegmark2025}, define the effective temperature:
\begin{equation}
    T_{\mathrm{eff}} = \frac{\eta\,\mathrm{tr}(\Sigma_g)}{2BN} = \frac{\eta\,\sigma^2_{\mathrm{grad}}}{2B}
\end{equation}
where $\sigma^2_{\mathrm{grad}} = \mathrm{tr}(\Sigma_g)/N$ is the per-parameter gradient noise variance.

\subsection{Stationary Distribution of Singular Values}

For a single weight layer $W \in \R^{m \times n}$, the Langevin dynamics in the SVD basis $W = U\Sigma V^\top$ induces a distribution over singular values. In the stationary state and under the adiabatic approximation (rotations $U$, $V$ equilibrate faster than $\Sigma$):
\begin{equation}
    P_{\mathrm{ss}}(\{\sigma_i\}) \;\propto\; \prod_{i < j}|\sigma_i^2 - \sigma_j^2| \;\cdot\; \prod_i \sigma_i^{|m-n|} \;\cdot\; \exp\!\left(-\frac{U_{\mathrm{eff}}(\{\sigma_i\})}{T_{\mathrm{eff}}}\right)
\end{equation}
where the first product is the Vandermonde repulsion (from the Jacobian of the SVD parameterization), and:
\begin{equation}
    U_{\mathrm{eff}}(\{\sigma_i\}) = \sum_i \left[\frac{\lambda}{2}\sigma_i^2 - S_i(\sigma_i)\right]
\end{equation}
with $S_i(\sigma_i)$ encoding the signal contribution from data (how much the $i$-th singular direction reduces loss).

\paragraph{Tail behavior.} For small singular values (large $i$), the signal term $S_i \to 0$ (these directions encode noise, not features), and the distribution reduces to:
\begin{equation}
    P(\sigma) \;\sim\; \sigma^{|m-n|} \exp\!\left(-\frac{\lambda\sigma^2}{2T_{\mathrm{eff}}}\right), \qquad \text{(tail, no signal)}
\end{equation}
which for eigenvalues $\lambda_i = \sigma_i^2$ gives:
\begin{equation}
    P(\lambda) \;\sim\; \lambda^{(|m-n|-1)/2} \exp\!\left(-\frac{\lambda_{\mathrm{wd}}\,\lambda}{2T_{\mathrm{eff}}}\right).
\end{equation}

\paragraph{Heavy-tail emergence.} When strong signals are present ($S_i \gg 0$ for the top directions), they pull those singular values far from the bulk, creating the heavy tail. The effective exponent $\alpha$ of the resulting power-law-like distribution is determined by the ratio of signal strength to thermal noise:
\begin{equation}
    \alpha \;\approx\; 1 + \frac{\lambda}{T_{\mathrm{eff}}} \cdot f\!\left(\frac{\|g_{\mathrm{signal}}\|}{\sigma_{\mathrm{noise}} / \sqrt{B}}\right)
\end{equation}
where $f$ is a monotonically increasing function. This explains:
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item More data (higher SNR) $\to$ lower $\alpha$ (heavier tail)
    \item Higher LR (higher $T_{\mathrm{eff}}$) $\to$ higher $\alpha$ (more random)
    \item Higher weight decay $\to$ can go either way (compresses tail but also regularizes)
\end{itemize}

\subsection{$\alpha$ Reversal: Phase Transition Analysis}

Define the normalized order parameter $\varphi \in [0, 1]$:
\begin{equation}
    \varphi = \frac{\alpha_{\mathrm{rand}} - \alpha}{\alpha_{\mathrm{rand}} - \alpha_{\mathrm{opt}}} \approx \frac{18 - \alpha}{18 - 2} = \frac{18 - \alpha}{16}.
\end{equation}

The dynamics of $\varphi$ follow a Landau-like relaxation:
\begin{equation}
    \gamma\frac{d\varphi}{dt} = h(t) - r\,\varphi - u\,\varphi^3
    \label{eq:landau_dynamics}
\end{equation}
where:
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item $h(t) \geq 0$: external field from data signal. Decreases as data information is absorbed.
    \item $r = r_0(T_{\mathrm{eff}} - T_c(N))$: mass parameter. Positive when $T_{\mathrm{eff}} > T_c(N)$ (disorder favored).
    \item $T_c(N)$: critical temperature, hypothesized to \emph{decrease} with $N$ (larger models are more fragile).
    \item $u > 0$: stabilizing quartic term.
\end{itemize}

\paragraph{Three phases from Equation~\ref{eq:landau_dynamics}:}
\begin{enumerate}[leftmargin=*,itemsep=1pt]
    \item \textbf{Phase 1 (Explosive ordering)}: $h(t)$ dominates; $d\varphi/dt > 0$ regardless of $r$. Duration: until data information begins to saturate.
    \item \textbf{Phase 2 (Reversal)}: $h(t) \to 0$; if $r > 0$ (which requires $T_{\mathrm{eff}} > T_c(N)$), the equilibrium shifts to $\varphi^* = 0$. The system relaxes back: $\varphi(t) \sim \varphi_0 \exp(-rt/\gamma)$, i.e., $\alpha$ increases.
    \item \textbf{Phase 3 (Recovery)}: If training continues long enough with fresh data, $h(t)$ accumulates slowly, eventually overcoming the disorder tendency. Recovery requires $\int h(t)\,dt > r\,\varphi\,T_{\mathrm{total}}$.
\end{enumerate}

\paragraph{Prediction: $T_c(N)$ scales with model size.}
If $T_c(N) \propto N^{-\gamma}$ for some $\gamma > 0$, then at fixed $T_{\mathrm{eff}}$ (fixed LR/batch):
\begin{equation}
    r(N) = r_0\bigl(T_{\mathrm{eff}} - T_c(N)\bigr) \;\approx\; r_0\,T_{\mathrm{eff}} \quad \text{for large } N
\end{equation}
meaning the reversal rate $d\alpha/dt \propto r \propto T_{\mathrm{eff}}$ is approximately constant across scales, but the \emph{duration} of Phase~2 grows with training length, producing larger $\Delta\alpha$ for larger models trained on more tokens. OLMo-2-13B's extreme reversal ($\Delta\alpha = 2.71$ over 590K steps) vs.\ Pythia-2.8B ($\Delta\alpha = 0.45$ over 133K steps) is consistent with this scaling.

\section{Randomized SVD Implementation}
\label{app:svd}
%=============================================================================

For weight matrices with $\min(m,n) > 1024$, we use randomized SVD~\citep{halko2011}:

\begin{algorithm}[t]
\caption{Spectral metric computation for one weight layer}
\label{alg:svd}
\begin{algorithmic}[1]
\REQUIRE Weight matrix $W \in \R^{m \times n}$, rank $k = 256$, oversampling $p = 16$
\STATE $\Omega \leftarrow \mathrm{randn}(n, k+p)$ \COMMENT{Random projection}
\STATE $Y \leftarrow W\Omega$; \quad $Q, \_ \leftarrow \mathrm{QR}(Y)$
\STATE $Q \leftarrow \mathrm{orth}(W(W^\top Q))$ \COMMENT{One power iteration}
\STATE $B \leftarrow Q^\top W$; \quad $\_, \Sigma, \_ \leftarrow \mathrm{SVD}(B)$
\STATE $\sigma_1, \ldots, \sigma_k \leftarrow$ top-$k$ singular values from $\Sigma$
\STATE $\mathrm{SR} \leftarrow \|W\|_F^2 / \sigma_1^2$ \COMMENT{Only needs Frobenius norm + $\sigma_1$}
\STATE $p_i \leftarrow \sigma_i^2 / \sum_j \sigma_j^2$
\STATE Fit $\alpha$ from $\{p_i\}$ using MLE + KS method
\STATE $C_{10} \leftarrow \sum_{i=1}^{10} p_i$
\RETURN SR, $\alpha$, $C_{10}$
\end{algorithmic}
\end{algorithm}

Cost: $O(mn(k+p))$ per layer. For a 13B model (${\sim}120$ weight layers, largest $5120 \times 20480$), total measurement time is ${\sim}23$ minutes on 8 GPUs with model sharding.

\paragraph{Validation.} On 50 layers from Pythia-1B, randomized SVD (top-256) produces $\alpha$ within 3\% and SR within 0.5\% of full SVD values. These approximation errors are negligible compared to inter-checkpoint variations.

\section{Power-Law Fitting Procedure}
\label{app:alpha_fit}
%=============================================================================

We fit $\alpha$ using the method of~\citet{martin_mahoney_jmlr2021}:

\begin{enumerate}[leftmargin=*,itemsep=1pt]
    \item Compute eigenvalues $\lambda_i = \sigma_i^2$ from the top-$k$ SVD.
    \item Select the fitting range: $[\lambda_{\min}, \lambda_{\max}]$ where $\lambda_{\min}$ is chosen by KS-statistic minimization and $\lambda_{\max} = \lambda_1$.
    \item Fit power-law exponent $\alpha$ by maximum likelihood: $\hat\alpha = 1 + n \left[\sum_{i=1}^n \log\frac{\lambda_i}{\lambda_{\min}}\right]^{-1}$.
    \item Report the per-layer $\alpha$ and compute the parameter-weighted mean across all 2D layers.
\end{enumerate}

We additionally compute a fast approximation via log-log linear regression on rank-ordered eigenvalues (used for the $\alpha$-guided schedule's online monitoring, where speed matters more than precision). The two methods produce Pearson $r > 0.97$ correlation across checkpoints, with the regression method tending to slightly lower $\alpha$ values.

\subsection{$\alpha$-Guided Schedule Implementation}

\begin{verbatim}
def alpha_guided_lr(step, cfg, alpha_history):
    if step < cfg.warmup_steps:
        return cfg.peak_lr * step / cfg.warmup_steps

    # Check for reversal: 3 consecutive increases
    if len(alpha_history) >= 4:
        recent = alpha_history[-4:]
        if all(recent[i+1] > recent[i] for i in range(3)):
            decay_start = step

    # Fallback: force decay at 80% if no reversal
    if step > 0.8 * cfg.total_steps:
        decay_start = step

    if decay_start is not None:
        remaining = cfg.total_steps - decay_start
        progress = (step - decay_start) / remaining
        return cfg.peak_lr - progress * (cfg.peak_lr - cfg.min_lr)

    return cfg.peak_lr  # Constant LR phase
\end{verbatim}

$\alpha$ is measured every 500 steps using fast SVD on 4 representative layers (${\sim}5$ seconds overhead per measurement for a 410M model).

\subsection{Per-Model $\alpha$ Trajectories}

The full $\alpha$ trajectory for each Pythia model shows a clear pattern:
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item \textbf{70M--410M} ($D/N > 700$): $\alpha$ monotonically decreases throughout training, reaching $\alpha < 3$ (heavy-tail regime).
    \item \textbf{1B} ($D/N = 297$): $\alpha$ reaches minimum at ${\sim}2.52$, then slowly rises to 2.78 (marginal reversal, $\Delta\alpha = 0.26$).
    \item \textbf{2.8B--6.9B} ($D/N < 110$): $\alpha$ reaches minimum at ${\sim}4.7$ by step 10--20K, then rises to ${\sim}5.1$--$5.2$ (clear reversal, $\Delta\alpha \approx 0.4$).
    \item \textbf{OLMo-2-13B} ($D/N = 365$): $\alpha$ reaches minimum 4.25 at step 7K, then rises continuously to 6.95 by step 596K ($\Delta\alpha = 2.71$, largest observed).
\end{itemize}
